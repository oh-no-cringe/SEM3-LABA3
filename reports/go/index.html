func (c *console) array() error {
        var a structures.Array
        fmt.Fprintln(c.out, "PUSH значение | INSERT индекс значение | GET индекс | SET индекс значение\nDEL индекс | SIZE | EMPTY | PRINT | EXIT")
        for {
                command, err := c.word()
                if err == io.EOF {
                        return nil
                }
                if err != nil {
                        return err
                }
                if command == "SAVE" || command == "LOAD" {
                        if err := c.file(command, &a, "array"); err != nil {
                                return err
                        }
                        continue
                }
                switch command {
                case "EXIT":
                        return nil
                case "PUSH":
                        v, err := c.word()
                        if err != nil {
                                return err
                        }
                        a.PushBack(v)
                case "INSERT", "SET":
                        i, err := c.index()
                        if err != nil {
                                return err
                        }
                        v, err := c.word()
                        if err != nil {
                                return err
                        }
                        if command == "INSERT" {
                                err = a.PushIndex(v, i)
                        } else {
                                err = a.Set(i, v)
                        }
                        if err != nil {
                                return err
                        }
                case "GET":
                        i, err := c.index()
                        if err != nil {
                                return err
                        }
                        v, err := a.Get(i)
                        if err != nil {
                                return err
                        }
                        fmt.Fprintln(c.out, v)
                        continue
                case "DEL":
                        i, err := c.index()
                        if err != nil {
                                return err
                        }
                        a.PopIndex(i)
                case "SIZE":
                        fmt.Fprintln(c.out, a.Length())
                        continue
                case "EMPTY":
                        c.printBool(a.Empty())
                        continue
                case "PRINT":
                default:
                        return fmt.Errorf("неизвестная команда массива")
                }
                a.Print(c.out)
        }
}

func (c *console) hash() error {

        method, err := c.number()
        if err != nil {
                return err
        }
        if method != 1 && method != 2 {
                return fmt.Errorf("нужен метод 1 или 2")
        }
        table := structures.NewHash(method == 2)
        fmt.Fprintln(c.out, "SET ключ значение | GET ключ | HAS ключ | DEL ключ | PRINT | EXIT")
        for {
                command, err := c.word()
                if err == io.EOF {
                        return nil
                }
                if err != nil {
                        return err
                }
                if command == "SAVE" || command == "LOAD" {
                        if err := c.file(command, table, "hash"); err != nil {
                                return err
                        }
                        continue
                }
                if command == "EXIT" {
                        return nil
                }
                if command == "PRINT" {
                        table.Print(c.out)
                        continue
                }
                if command != "SET" && command != "GET" && command != "HAS" && command != "DEL" {
                        return fmt.Errorf("неизвестная команда хеш-таблицы")
                }
                key, err := c.word()
                if err != nil {
                        return err
                }
                switch command {
                case "SET":
                        value, err := c.word()
                        if err != nil {
                                return err
                        }
                        table.Set(key, value)
                case "GET":
                        value, ok := table.Get(key)
                        if ok {
                                fmt.Fprintln(c.out, value)
                        } else {
                                fmt.Fprintln(c.out, "NOT_FOUND")
                        }
                        continue
                case "HAS":
                        c.printBool(table.Has(key))
                        continue
                case "DEL":
                        table.Del(key)
                }

        }

type console struct {
        in  *bufio.Reader
        out io.Writer
}

func space(c byte) bool { return strings.ContainsRune(" \t\r\n\v\f", rune(c)) }
func (c *console) word() (string, error) {
        var first byte
        for {
                b, err := c.in.ReadByte()
                if err != nil {
                        return "", err
                }
                if !space(b) {
                        first = b
                        break
                }
        }
        var text strings.Builder
        if first == '"' {
                for {
                        b, err := c.in.ReadByte()
                        if err != nil {
                                return "", fmt.Errorf("незакрытая строка")
                        }
                        if b == '"' {
                                next, err := c.in.Peek(1)
                                if err == nil && !space(next[0]) {
                                        return "", fmt.Errorf("между аргументами нужен пробел")
                                }
                                return text.String(), nil
                        }
                        if b == '\\' {
                                b, err = c.in.ReadByte()
                                if err != nil {
                                        return "", fmt.Errorf("незакрытая строка")
                                }
                        }
                        text.WriteByte(b)
                }
        }
        text.WriteByte(first)
        for {
                b, err := c.in.ReadByte()
                if err == io.EOF {
                        return text.String(), nil
                }
                if err != nil {
                        return "", err
                }
                if space(b) {
                        return text.String(), nil
                }
                text.WriteByte(b)
        }
}
func (c *console) number() (int, error) {
        s, err := c.word()
        if err != nil {
                return 0, err
        }
        n, err := strconv.ParseInt(s, 10, 32)
        if err != nil {
                return 0, fmt.Errorf("нужно целое число")
        }
        return int(n), nil
}
func (c *console) index() (int, error) {
        n, err := c.number()
        if err != nil {
                return 0, err
        }
        if n < 0 {
                return 0, fmt.Errorf("индекс должен быть неотрицательным")
        }
        return n, nil
}
func (c *console) printBool(b bool) {
        if b {
                fmt.Fprintln(c.out, "TRUE")
        } else {
                fmt.Fprintln(c.out, "FALSE")
        }
}

                return err

        case 1:

        case 2:
                var l structures.ForwardList
                return c.list(&l)
        case 3:
                var l structures.List
                return c.list(&l)

        case 6:
                return c.hash()
        case 7:
                return c.tree()
        default:
                return fmt.Errorf("нужен номер от 1 до 7")
        }
}

func (c *console) stackQueue(data stackQueueMethods, kind string) error {

        for {
                command, err := c.word()
                if err == io.EOF {
                        return nil
                }

                        return err
                }

                                return err

                }
                switch command {
                case "EXIT":
                        return nil
                case "PUSH":
                        v, err := c.word()
                        if err != nil {
                                return err
                        }
                        data.Push(v)
                case "POP":
                        if data.Empty() {
                                fmt.Fprintln(c.out, "EMPTY")
                        } else {
                                v, err := data.Pop()
                                if err != nil {
                                        return err
                                }
                                fmt.Fprintln(c.out, v)
                        }
                        continue
                case "EMPTY":
                        c.printBool(data.Empty())
                        continue
                case "PRINT":
                default:
                        return fmt.Errorf("неизвестная команда стека или очереди")
                }
                data.Print(c.out)
        }
}

func (c *console) file(command string, object any, kind string) error {
        format, err := c.word()
        if err != nil {
                return err
        }
        path, err := c.word()
        if err != nil {
                return err
        }
        if format != "text" && format != "binary" {
                return fmt.Errorf("формат: text или binary")
        }
        if command == "SAVE" {
                err = structures.SaveFile(object, path, kind, format == "binary")
        } else {
                err = structures.LoadFile(object, path, kind, format == "binary")
        }
        if err == nil {
                fmt.Fprintln(c.out, "OK")
        }
        return err
}

type Array struct {
        data []string
        size int
}

func (a *Array) reserve() {
        if a.size < len(a.data) {
                return
        }
        capacity := len(a.data) * 2
        if capacity == 0 {
                capacity = 4
        }
        next := make([]string, capacity)
        copy(next, a.data)
        a.data = next
}
func (a *Array) PushBack(value string) {
        a.reserve()
        a.data[a.size] = value
        a.size++
}
func (a *Array) PushIndex(value string, index int) error {
        if index < 0 || index > a.size {
                return fmt.Errorf("array index out of range")
        }
        a.reserve()
        copy(a.data[index+1:a.size+1], a.data[index:a.size])
        a.data[index] = value
        a.size++
        return nil
}
func (a *Array) Get(index int) (string, error) {
        if index < 0 || index >= a.size {
                return "", fmt.Errorf("array index out of range")
        }
        return a.data[index], nil
}
func (a *Array) Set(index int, value string) error {
        if index < 0 || index >= a.size {
                return fmt.Errorf("array index out of range")
        }
        a.data[index] = value
        return nil
}
func (a *Array) PopIndex(index int) {
        if index < 0 || index >= a.size {
                return
        }
        copy(a.data[index:a.size-1], a.data[index+1:a.size])
        a.size--
        a.data[a.size] = ""
}
func (a *Array) Length() int {
        return a.size
}
func (a *Array) Empty() bool {
        return a.size == 0
}
func (a *Array) Values() []string {
        return append([]string{}, a.data[:a.size]...)
}

func (a *Array) Clear() {
        for i := 0; i < a.size; i++ {
                a.data[i] = ""
        }
        a.size = 0
}

func (a *Array) Swap(other *Array) {
        a.data, other.data = other.data, a.data
        a.size, other.size = other.size, a.size
}

type forwardNode struct {
        value string
        next  *forwardNode
}
type ForwardList struct{ head *forwardNode }

func (l *ForwardList) PushFront(v string) {
        l.head = &forwardNode{v, l.head}
}
func (l *ForwardList) PushBack(v string) {
        n := &forwardNode{value: v}
        if l.head == nil {
                l.head = n
                return
        }
        c := l.head
        for c.next != nil {
                c = c.next
        }
        c.next = n
}
func (l *ForwardList) PushIndex(v string, index int) {
        if index < 0 {
                return
        }
        if index == 0 {
                l.PushFront(v)
                return
        }
        c := l.head
        for i := 0; i < index-1 && c != nil; i++ {
                c = c.next
        }
        if c != nil {
                c.next = &forwardNode{v, c.next}
        }
}
func (l *ForwardList) PopFront() {
        if l.head != nil {
                l.head = l.head.next
        }
}
func (l *ForwardList) PopBack() {
        if l.head == nil {
                return
        }
        if l.head.next == nil {
                l.head = nil
                return
        }
        c := l.head
        for c.next.next != nil {
                c = c.next
        }
        c.next = nil
}
func (l *ForwardList) PopIndex(index int) {
        if index < 0 {
                return
        }
        if index == 0 {
                l.PopFront()
                return
        }
        c := l.head
        for i := 0; i < index-1 && c != nil; i++ {
                c = c.next
        }
        if c != nil && c.next != nil {
                c.next = c.next.next
        }
}
func (l *ForwardList) Find(v string) bool {
        for c := l.head; c != nil; c = c.next {
                if c.value == v {
                        return true
                }
        }
        return false
}
func (l *ForwardList) InsertBefore(target, v string) bool {
        var prev *forwardNode
        c := l.head
        for c != nil && c.value != target {
                prev = c
                c = c.next
        }
        if c == nil {
                return false
        }
        n := &forwardNode{v, c}
        if prev == nil {
                l.head = n
        } else {
                prev.next = n
        }
        return true
}
func (l *ForwardList) InsertAfter(target, v string) bool {
        c := l.head
        for c != nil && c.value != target {
                c = c.next
        }
        if c == nil {
                return false
        }
        c.next = &forwardNode{v, c.next}
        return true
}
func (l *ForwardList) EraseValue(v string) bool {
        var prev *forwardNode
        c := l.head
        for c != nil && c.value != v {
                prev = c
                c = c.next
        }
        if c == nil {
                return false
        }
        if prev == nil {
                l.head = c.next
        } else {
                prev.next = c.next
        }
        return true
}
func (l *ForwardList) EraseBefore(target string) bool {
        var before, prev *forwardNode
        c := l.head
        for c != nil && c.value != target {
                before = prev
                prev = c
                c = c.next
        }
        if c == nil || prev == nil {
                return false
        }
        if before == nil {
                l.head = c
        } else {
                before.next = c
        }
        return true
}
func (l *ForwardList) EraseAfter(target string) bool {
        c := l.head
        for c != nil && c.value != target {
                c = c.next
        }
        if c == nil || c.next == nil {
                return false
        }
        c.next = c.next.next
        return true
}
func (l *ForwardList) Empty() bool {
        return l.head == nil
}
func (l *ForwardList) Values() []string {
        v := []string{}
        for c := l.head; c != nil; c = c.next {
                v = append(v, c.value)
        }
        return v
}

type HashTable struct {
        keys, data  Array
        count, used int
        double      bool
}

func NewHash(double bool) *HashTable {
        table := &HashTable{double: double}
        for i := 0; i < 8; i++ {
                table.keys.PushBack("")
                table.data.PushBack("")
        }
        return table
}

func (t *HashTable) hashStep(key string) int {
        if !t.double {
                return 1
        }
        var number uint64
        for i := 0; i < len(key); i++ {
                number = number*131 + uint64(key[i])
        }
        // Нечётный шаг проходит все ячейки при размере 8, 16, 32 и далее.
        return 2*int(number%uint64(t.keys.Length()/2)) + 1
}

func (t *HashTable) findPos(key string) int {
        size := t.keys.Length()
        // FNV-1a вычисляет начальную позицию в обоих режимах.
        var hash uint64 = 14695981039346656037
        for i := 0; i < len(key); i++ {
                hash ^= uint64(key[i])
                hash *= 1099511628211
        }
        pos := int(hash % uint64(size))
        jump := t.hashStep(key)
        deleted := size
        wanted := "+" + key

        for i := 0; i < size; i++ {
                cell := t.keys.data[pos]
                if cell == wanted {
                        return pos
                }
                if cell == "" {
                        if deleted != size {
                                return deleted
                        }
                        return pos
                }
                if cell == "-" && deleted == size {
                        deleted = pos
                }
                pos = (pos + jump) % size
        }
        return deleted
}

func (t *HashTable) Set(key, value string) {
        pos := t.findPos(key)
        size := t.keys.Length()
        if pos < size && t.keys.data[pos] == "+"+key {
                t.data.data[pos] = value
                return
        }
        if t.count >= size/2 || pos == size {
                t.rebuild(true)
                pos = t.findPos(key)
        } else if t.used >= size-size/4 && t.keys.data[pos] == "" {
                t.rebuild(false)
                pos = t.findPos(key)
        }

        if t.keys.data[pos] == "" {
                t.used++
        }
        t.keys.data[pos] = "+" + key
        t.data.data[pos] = value
        t.count++
}

func (t *HashTable) Get(key string) (string, bool) {
        pos := t.findPos(key)
        if pos == t.keys.Length() || t.keys.data[pos] != "+"+key {
                return "", false
        }
        return t.data.data[pos], true
}

func (t *HashTable) Has(key string) bool {
        _, found := t.Get(key)
        return found
}

func (t *HashTable) Del(key string) {
        pos := t.findPos(key)
        if pos < t.keys.Length() && t.keys.data[pos] == "+"+key {
                t.keys.data[pos] = "-"
                t.data.data[pos] = ""
                t.count--
        }
}

func (t *HashTable) Names() []string {
        names := []string{}
        for i := 0; i < t.keys.Length(); i++ {
                key := t.keys.data[i]
                if len(key) > 0 && key[0] == '+' {
                        names = append(names, key[1:])
                }
        }
        return names
}

func (t *HashTable) rebuild(grow bool) {
        var keys, values Array
        size := t.keys.Length()
        for i := 0; i < size; i++ {
                cell := t.keys.data[i]
                if len(cell) > 0 && cell[0] == '+' {
                        keys.PushBack(cell[1:])
                        values.PushBack(t.data.data[i])
                }
                t.keys.data[i] = ""
                t.data.data[i] = ""
        }
        if grow {
                for i := 0; i < size; i++ {
                        t.keys.PushBack("")
                        t.data.PushBack("")
                }
        }
        t.count = 0
        t.used = 0
        for i := 0; i < keys.Length(); i++ {
                t.Set(keys.data[i], values.data[i])
        }
}

type listNode struct {
        value      string
        next, prev *listNode
}
type List struct{ head, tail *listNode }

func (l *List) PushFront(v string) {
        n := &listNode{value: v, next: l.head}
        if l.head != nil {
                l.head.prev = n
        } else {
                l.tail = n
        }
        l.head = n
}
func (l *List) PushBack(v string) {
        n := &listNode{value: v, prev: l.tail}
        if l.tail != nil {
                l.tail.next = n
        } else {
                l.head = n
        }
        l.tail = n
}
func (l *List) PushIndex(v string, index int) {
        if index < 0 {
                return
        }
        if index == 0 {
                l.PushFront(v)
                return
        }
        c := l.head
        for i := 0; i < index-1 && c != nil; i++ {
                c = c.next
        }
        if c == nil {
                l.PushBack(v)
                return
        }
        l.insertAfterNode(c, v)
}
func (l *List) unlink(c *listNode) {
        if c.prev != nil {
                c.prev.next = c.next
        } else {
                l.head = c.next
        }
        if c.next != nil {
                c.next.prev = c.prev
        } else {
                l.tail = c.prev
        }
        c.next = nil
        c.prev = nil
}
func (l *List) PopFront() {
        if l.head != nil {
                l.unlink(l.head)
        }
}
func (l *List) PopBack() {
        if l.tail != nil {
                l.unlink(l.tail)
        }
}
func (l *List) PopIndex(index int) {
        if index < 0 {
                return
        }
        c := l.head
        for i := 0; i < index && c != nil; i++ {
                c = c.next
        }
        if c != nil {
                l.unlink(c)
        }
}
func (l *List) findNode(v string) *listNode {
        for c := l.head; c != nil; c = c.next {
                if c.value == v {
                        return c
                }
        }
        return nil
}
func (l *List) Find(v string) bool {
        return l.findNode(v) != nil
}
func (l *List) InsertBefore(target, v string) bool {
        c := l.findNode(target)
        if c == nil {
                return false
        }
        n := &listNode{value: v, next: c, prev: c.prev}
        if c.prev != nil {
                c.prev.next = n
        } else {
                l.head = n
        }
        c.prev = n
        return true
}
func (l *List) insertAfterNode(c *listNode, v string) {
        n := &listNode{value: v, next: c.next, prev: c}
        if c.next != nil {
                c.next.prev = n
        } else {
                l.tail = n
        }
        c.next = n
}
func (l *List) InsertAfter(target, v string) bool {
        c := l.findNode(target)
        if c == nil {
                return false
        }
        l.insertAfterNode(c, v)
        return true
}
func (l *List) EraseValue(v string) bool {
        c := l.findNode(v)
        if c == nil {
                return false
        }
        l.unlink(c)
        return true
}
func (l *List) EraseBefore(target string) bool {
        c := l.findNode(target)
        if c == nil || c.prev == nil {
                return false
        }
        l.unlink(c.prev)
        return true
}
func (l *List) EraseAfter(target string) bool {
        c := l.findNode(target)
        if c == nil || c.next == nil {
                return false
        }
        l.unlink(c.next)
        return true
}
func (l *List) Empty() bool {
        return l.head == nil
}
func (l *List) Values() []string {
        v := []string{}
        for c := l.head; c != nil; c = c.next {
                v = append(v, c.value)
        }
        return v
}

type queueNode[T any] struct {
        value T
        next  *queueNode[T]
}

type Queue[T any] struct {
        head *queueNode[T]
        tail *queueNode[T]
}

func (q *Queue[T]) Push(value T) {
        newNode := &queueNode[T]{value: value}
        if q.tail == nil {
                q.head = newNode
        } else {
                q.tail.next = newNode
        }
        q.tail = newNode
}

func (q *Queue[T]) Pop() (T, error) {
        if q.head == nil {
                var value T
                return value, fmt.Errorf("очередь пуста")
        }
        temp := q.head
        q.head = temp.next
        if q.head == nil {
                q.tail = nil
        }
        return temp.value, nil
}

func (q *Queue[T]) Empty() bool {
        return q.head == nil
}

func (q *Queue[T]) Values() []T {
        values := []T{}
        for node := q.head; node != nil; node = node.next {
                values = append(values, node.value)
        }
        return values
}

func writeNumber(w io.Writer, n uint32, bin bool) error {
        if bin {
                return binary.Write(w, binary.LittleEndian, n)
        }
        _, err := fmt.Fprintln(w, n)
        return err
}
func readNumber(r *bufio.Reader, bin bool) (uint32, error) {
        var n uint32
        if bin {
                err := binary.Read(r, binary.LittleEndian, &n)
                return n, err
        }
        s, err := r.ReadString('\n')
        if err != nil {
                return 0, err
        }
        s = strings.TrimSuffix(s, "\n")
        if s == "" || strings.IndexFunc(s, func(c rune) bool { return c < '0' || c > '9' }) >= 0 {
                return 0, fmt.Errorf("неверное число")
        }
        value, err := strconv.ParseUint(s, 10, 32)
        return uint32(value), err
}
func writeString(w io.Writer, s string, bin bool) error {
        if len(s) > 1048576 {
                return fmt.Errorf("слишком длинная строка")
        }
        if err := writeNumber(w, uint32(len(s)), bin); err != nil {
                return err
        }
        if _, err := io.WriteString(w, s); err != nil {
                return err
        }
        if !bin {
                _, err := io.WriteString(w, "\n")
                return err
        }
        return nil
}
func readString(r *bufio.Reader, bin bool) (string, error) {
        n, err := readNumber(r, bin)
        if err != nil {
                return "", err
        }
        if n > 1048576 {
                return "", fmt.Errorf("слишком длинная строка")
        }
        data := make([]byte, n)
        if _, err = io.ReadFull(r, data); err != nil {
                return "", err
        }
        if !bin {
                c, err := r.ReadByte()
                if err != nil || c != '\n' {
                        return "", fmt.Errorf("нет разделителя")
                }
        }
        return string(data), nil
}
func snapshot(object any) ([]string, error) {
        switch o := object.(type) {
        case *Array:
                return o.Values(), nil
        case *ForwardList:
                return o.Values(), nil
        case *List:
                return o.Values(), nil
        case *Stack:
                return o.Values(), nil
        case *Queue[string]:
                return o.Values(), nil
        case *Tree:
                return o.Values(), nil
        case *HashTable:
                mode := "0"
                if o.double {
                        mode = "1"
                }
                v := []string{mode}
                for _, key := range o.Names() {
                        value, _ := o.Get(key)
                        v = append(v, key, value)
                }
                return v, nil
        default:
                return nil, fmt.Errorf("неизвестная структура")
        }
}
func restore(object any, v []string) error {
        switch o := object.(type) {
        case *Array:
                o.Clear()
                for _, s := range v {
                        o.PushBack(s)
                }
        case *ForwardList:
                *o = ForwardList{}
                for _, s := range v {
                        o.PushBack(s)
                }
        case *List:
                *o = List{}
                for _, s := range v {
                        o.PushBack(s)
                }
        case *Stack:
                *o = Stack{}
                for i := len(v); i > 0; i-- {
                        o.Push(v[i-1])
                }
        case *Queue[string]:
                *o = Queue[string]{}
                for _, s := range v {
                        o.Push(s)
                }
        case *Tree:
                var temp Tree
                for _, s := range v {
                        n, err := strconv.Atoi(s)
                        if err != nil {
                                return err
                        }
                        temp.Insert(n)
                }
                *o = temp
        case *HashTable:
                if len(v) == 0 || len(v)%2 == 0 || (v[0] != "0" && v[0] != "1") {
                        return fmt.Errorf("неверная таблица")
                }
                temp := NewHash(v[0] == "1")
                for i := 1; i < len(v); i += 2 {
                        if temp.Has(v[i]) {
                                return fmt.Errorf("дубликат ключа")
                        }
                        temp.Set(v[i], v[i+1])
                }
                *o = *temp
        default:
                return fmt.Errorf("неизвестная структура")
        }
        return nil
}

// SaveFile сохраняет порядок элементов и режим хеширования в общий формат C++/Go.
func SaveFile(object any, path, kind string, bin bool) error {
        values, err := snapshot(object)
        if err != nil {
                return err
        }
        if len(values) > 100000 {
                return fmt.Errorf("слишком много элементов")
        }
        f, err := os.Create(path)
        if err != nil {
                return err
        }
        defer f.Close()
        header := "LR3TEXT1\n"
        if bin {
                header = "LR3BIN1\n"
        }
        if _, err = io.WriteString(f, header); err != nil {
                return err
        }
        if err = writeString(f, kind, bin); err != nil {
                return err
        }
        if err = writeNumber(f, uint32(len(values)), bin); err != nil {
                return err
        }
        for _, s := range values {
                if err = writeString(f, s, bin); err != nil {
                        return err
                }
        }
        return f.Close()
}

// LoadFile сначала читает и проверяет файл, затем восстанавливает структуру.
func LoadFile(object any, path, kind string, bin bool) error {
        f, err := os.Open(path)
        if err != nil {
                return err
        }
        defer f.Close()
        r := bufio.NewReader(f)
        header, err := r.ReadString('\n')
        want := "LR3TEXT1\n"
        if bin {
                want = "LR3BIN1\n"
        }
        if err != nil || header != want {
                return fmt.Errorf("неверный формат")
        }
        tag, err := readString(r, bin)
        if err != nil {
                return err
        }
        if tag != kind {
                return fmt.Errorf("другой тип структуры")
        }
        n, err := readNumber(r, bin)
        if err != nil {
                return err
        }
        if n > 100000 {
                return fmt.Errorf("слишком много элементов")
        }
        values := []string{}
        for i := uint32(0); i < n; i++ {
                s, err := readString(r, bin)
                if err != nil {
                        return err
                }
                values = append(values, s)
        }
        if _, err = r.ReadByte(); err != io.EOF {
                return fmt.Errorf("лишние данные")
        }
        return restore(object, values)
}

import "strconv"

type treeNode struct {
        value int
        left  *treeNode
        right *treeNode
}

type Tree struct {
        root *treeNode
}

func (t *Tree) Empty() bool {
        return t.root == nil
}

func (t *Tree) Insert(value int) {
        newNode := &treeNode{value: value}
        if t.root == nil {
                t.root = newNode
                return
        }
        var next Queue[*treeNode]
        next.Push(t.root)
        for !next.Empty() {
                node, _ := next.Pop()
                if node.left == nil {
                        node.left = newNode
                        return
                }
                if node.right == nil {
                        node.right = newNode
                        return
                }
                next.Push(node.left)
                next.Push(node.right)
        }
}

func findValue(node *treeNode, value int) bool {
        if node == nil {
                return false
        }
        if node.value == value {
                return true
        }
        return findValue(node.left, value) || findValue(node.right, value)
}

func (t *Tree) Search(value int) bool {
        return findValue(t.root, value)
}

func (t *Tree) IsComplete() bool {
        var next Queue[*treeNode]
        next.Push(t.root)
        gap := false
        for !next.Empty() {
                node, _ := next.Pop()
                if node == nil {
                        gap = true
                } else {
                        if gap {
                                return false
                        }
                        next.Push(node.left)
                        next.Push(node.right)
                }
        }
        return true
}

func (t *Tree) Values() []string {
        values := []string{}
        var next Queue[*treeNode]
        if t.root != nil {
                next.Push(t.root)
        }
        for !next.Empty() {
                node, _ := next.Pop()
                values = append(values, strconv.Itoa(node.value))
                if node.left != nil {
                        next.Push(node.left)
                }
                if node.right != nil {
                        next.Push(node.right)
                }
        }
        return values
}

func visit(node *treeNode, order string, values *[]string) {
        if node == nil {
                return
        }
        if order == "PRE" {
                *values = append(*values, strconv.Itoa(node.value))
        }
        visit(node.left, order, values)
        if order == "IN" {
                *values = append(*values, strconv.Itoa(node.value))
        }
        visit(node.right, order, values)
        if order == "POST" {
                *values = append(*values, strconv.Itoa(node.value))
        }
}

func (t *Tree) Traverse(order string) []string {
        values := []string{}
        visit(t.root, order, &values)
        return values
}
